# Inteligencia Objetiva 2.0 — reglas para servir la tienda nueva en las URL actuales del sitio.
# Se agregan al .htaccess de la RAÍZ del sitio, ANTES de las reglas existentes, solo cuando se apruebe el cambio (fase 5 de la guía).
# Para volver a las páginas anteriores basta con comentar este bloque.

RewriteEngine On
RewriteRule ^nosotros/?$                    /io/tienda/nosotros.php [L,QSA]
RewriteRule ^catalogo/?$                    /io/tienda/catalogo.php [L,QSA]
RewriteRule ^conferencias-en-linea/?$       /io/tienda/catalogo.php?categoria=conferencias [L,QSA]
RewriteRule ^cursos-en-linea/?$             /io/tienda/catalogo.php?categoria=cursos [L,QSA]
RewriteRule ^(documentales|introspecciones)/?$ /io/tienda/catalogo.php?categoria=documentales [L,QSA]
RewriteRule ^audiolibro/?$                  /io/tienda/catalogo.php?categoria=audiolibro [L,QSA]
RewriteRule ^producto/([a-z0-9-]+)/?$       /io/tienda/producto.php?p=$1 [L,QSA]
RewriteRule ^carrito/?$                     /io/tienda/carrito.php [L,QSA]
RewriteRule ^gracias/?$                     /io/tienda/gracias.php [L,QSA]
RewriteRule ^(en-vivo|calendario)/?$          /io/tienda/en-vivo.php [L,QSA]
RewriteRule ^membresias/?$                  /io/tienda/membresias.php [L,QSA]

# Buscadores y agentes de IA (sección 5.9 de la guía). Si el sitio ya tiene robots.txt o sitemap.xml, compáralos antes:
# estas reglas los reemplazan. Las líneas propias del sitio pueden conservarse en config.php → agentes.robots_extra.
RewriteRule ^robots\.txt$                  /io/tienda/agentes.php?archivo=robots [L]
RewriteRule ^sitemap\.xml$                 /io/tienda/agentes.php?archivo=sitemap [L]
RewriteRule ^llms\.txt$                    /io/tienda/agentes.php?archivo=llms [L]
RewriteRule ^llms-full\.txt$               /io/tienda/agentes.php?archivo=llms-full [L]

# Fichas antiguas (/conferencias-en-linea/titulo-410) → ficha nueva, con redirección permanente para SEO
RewriteRule ^(conferencias-en-linea|cursos-en-linea|documentales|introspecciones|audiolibro)/([a-z0-9-]+-[0-9]+)/?$ /producto/$2 [R=301,L]
